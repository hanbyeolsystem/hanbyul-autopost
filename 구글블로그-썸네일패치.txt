<script>//<![CDATA[
(function () {
  var phs = [].slice.call(document.querySelectorAll('a.post-thumb .post-thumb-ph'));
  if (!phs.length || !window.fetch) return;
  function firstImg(html) {
    var m = /<img[^>]+src=["']([^"']+)/i.exec(html || '');
    return m ? m[1] : '';
  }
  function key(u) { return (u || '').replace(/^https?:\/\/[^\/]+/, '').split(/[?#]/)[0]; }
  function fill(ph, src) {
    var a = ph.closest('a.post-thumb');
    if (!a || !src || a.querySelector('img')) return;
    var img = document.createElement('img');
    img.src = src; img.alt = ''; img.loading = 'lazy';
    img.style.cssText = 'position:absolute;inset:0;width:100%;height:100%;object-fit:cover;';
    img.onerror = function () { img.remove(); ph.style.display = ''; };
    a.appendChild(img);
    ph.style.display = 'none';
  }
  var lm = /\/search\/label\/([^\/?#]+)/.exec(location.pathname);
  var feed = lm ? '/feeds/posts/default/-/' + lm[1] + '?alt=json&max-results=150'
                : '/feeds/posts/default?alt=json&max-results=150';
  fetch(feed).then(function (r) { return r.json(); }).then(function (d) {
    var map = {};
    (d.feed.entry || []).forEach(function (e) {
      var href = '';
      (e.link || []).forEach(function (l) { if (l.rel === 'alternate') href = l.href; });
      var src = firstImg(e.content && e.content.$t);
      if (href && src) map[key(href)] = src;
    });
    return map;
  }).catch(function () { return {}; }).then(function (map) {
    var rest = [];
    phs.forEach(function (ph) {
      var a = ph.closest('a.post-thumb');
      var src = a && map[key(a.href)];
      if (src) fill(ph, src); else if (a) rest.push([ph, key(a.href)]);
    });
    rest.forEach(function (p) {
      fetch('/feeds/posts/default?alt=json&path=' + encodeURIComponent(p[1]))
        .then(function (r) { return r.json(); })
        .then(function (d) {
          var e = d.feed.entry && d.feed.entry[0];
          fill(p[0], firstImg(e && e.content && e.content.$t));
        }).catch(function () {});
    });
  });
})();
//]]></script>
